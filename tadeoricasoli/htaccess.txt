<IfModule mod_rewrite.c>
RewriteEngine On

# BLOQUEO USER-AGENT SOSPECHOSOS
RewriteCond %{HTTP_USER_AGENT} (bot|spider|crawler|fetch|scraper|ping|preview|curl|wget|Claude|GPT|OpenAI|BingPreview|LLaMA|Anthropic|Zyte|DataForSeo|Ahrefs|Semrush|MJ12bot|ScreamingFrog|Python-urllib|Java) [NC]
RewriteRule .* - [F,L]

# BLOQUEO REFERRERS SOSPECHOSOS
RewriteCond %{HTTP_REFERER} (free-money|buy-traffic|cheap-seo|malicious-site) [NC]
RewriteRule .* - [F,L]

# BLOQUEO POR HEADERS SOSPECHOSOS O INCOMPLETOS
RewriteCond %{HTTP_USER_AGENT} ^-?$
RewriteRule .* - [F,L]

# Evitar listado de directorios
Options -Indexes

# Bloquear ejecución de scripts en uploads
<FilesMatch "\.(php|pl|py|jsp|asp|sh|cgi)$">
  Order Deny,Allow
  Deny from all
</FilesMatch>
</IfModule>
